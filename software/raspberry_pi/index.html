架構調整重點：原本你的方塊圖中，APP 同時透過 Wi-Fi 連 Raspberry Pi、又透過 Bluetooth 連 Arduino。改成網頁控制後，瀏覽器無法穩定同時直連藍牙與 Wi-Fi，而且老師的 L02-H13 講義第 3～14 頁 正好教你用 Type-B USB 線連接 Raspberry Pi 與 Arduino Uno (`/dev/ttyACM0`)！因此最標準的做法是：瀏覽器全部透過 Wi-Fi 連到 Raspberry Pi 的網頁伺服器，再由 Raspberry Pi 透過 USB Serial 下達字元指令給 Arduino，這樣就不需要額外接藍牙模組，手機或電腦打開網址就能直接看畫面並操控車車。

Step 2 · Raspberry Pi Python 網頁控制後端 `rpi_web_server.py` (對應講義 L02-H13 第 8, 13 頁)

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Raspberry Pi 智慧吸塵車 Web 控制伺服器
整合 L02-H13 講義之 pyserial (/dev/ttyACM0, baudrate=115200)
"""
from http.server import SimpleHTTPRequestHandler, HTTPServer
import urllib.parse
import json
import serial
import threading
import time

SERIAL_PORT = '/dev/ttyACM0'  # 若為 ttyACM1 請依講義 L02-H13 p.8 修改
BAUD_RATE = 115200

latest_telemetry = {"distance": 50.0, "auto_stopped": False, "last_cmd": "S"}
ser = None

try:
    ser = serial.Serial(SERIAL_PORT, baudrate=BAUD_RATE, timeout=1.0)
    print(f"[OK] Connected to Arduino on {SERIAL_PORT}")
except Exception as e:
    print(f"[WARN] Serial not connected ({e}), running in dry-run mode.")

def read_from_arduino():
    """背景讀取 Arduino 傳回的超音波距離 (對應講義 L02-H13 p.8 readVal)"""
    global latest_telemetry
    while True:
        if ser and ser.is_open:
            try:
                line = ser.readline().decode('utf-8', errors='ignore').strip()
                if line.startswith("DIST:"):
                    dist = float(line.split(":")[1])
                    latest_telemetry["distance"] = dist
                    latest_telemetry["auto_stopped"] = (dist < 15.0)
            except Exception:
                pass
        time.sleep(0.05)

threading.Thread(target=read_from_arduino, daemon=True).start()

class CarRequestHandler(SimpleHTTPRequestHandler):
    def end_headers(self):
        self.send_header('Access-Control-Allow-Origin', '*')
        self.send_header('Access-Control-Allow-Methods', 'GET, POST, OPTIONS')
        self.send_header('Access-Control-Allow-Headers', 'Content-Type')
        super().end_headers()

    def do_OPTIONS(self):
        self.send_response(200)
        self.end_headers()

    def do_GET(self):
        parsed = urllib.parse.urlparse(self.path)
        if parsed.path == '/api/status':
            self.send_response(200)
            self.send_header('Content-type', 'application/json')
            self.end_headers()
            self.wfile.write(json.dumps(latest_telemetry).encode('utf-8'))
            return
        return super().do_GET()

    def do_POST(self):
        if self.path == '/api/command':
            length = int(self.headers.get('Content-Length', 0))
            body = json.loads(self.rfile.read(length).decode('utf-8') or "{}")
            cmd = str(body.get("cmd", "S"))[0]
            latest_telemetry["last_cmd"] = cmd
            if ser and ser.is_open:
                ser.write(cmd.encode('utf-8'))
            self.send_response(200)
            self.send_header('Content-type', 'application/json')
            self.end_headers()
            self.wfile.write(json.dumps({"status": "ok", "sent": cmd, "telemetry": latest_telemetry}).encode('utf-8'))

if __name__ == '__main__':
    server = HTTPServer(('0.0.0.0', 5000), CarRequestHandler)
    print("Web Control Server running at http://0.0.0.0:5000")
    server.serve_forever()

Step 3 · Arduino Uno 控制程式 `car_controller.ino` (對應講義 L02-H13 第 7, 12 頁 + 超音波自動停止)

#include <Servo.h>

// 腳位定義 (請依實際車車馬達驅動板 L298N / 感測器接線調整)
const int TRIG_PIN = 9;
const int ECHO_PIN = 10;
const int VACUUM_PIN = 7;     // 吸塵器馬達繼電器或 MOSFET 腳位
const int SERVO_PIN = 6;      // 伺服馬達腳位
const int IN1 = 2, IN2 = 3, IN3 = 4, IN4 = 5; // 驅動與轉向馬達腳位

const float SAFE_DISTANCE_CM = 15.0; // 超音波距離太近自動停止門檻
Servo myServo;
char currentMotion = 'S';

void setup() {
  Serial.begin(115200); // 對應講義 L02-H13 baudrate 115200
  pinMode(TRIG_PIN, OUTPUT);
  pinMode(ECHO_PIN, INPUT);
  pinMode(VACUUM_PIN, OUTPUT);
  pinMode(IN1, OUTPUT); pinMode(IN2, OUTPUT);
  pinMode(IN3, OUTPUT); pinMode(IN4, OUTPUT);
  myServo.attach(SERVO_PIN);
  myServo.write(90);
  stopMotors();
}

float readUltrasonicDistance() {
  digitalWrite(TRIG_PIN, LOW);
  delayMicroseconds(2);
  digitalWrite(TRIG_PIN, HIGH);
  delayMicroseconds(10);
  digitalWrite(TRIG_PIN, LOW);
  long duration = pulseIn(ECHO_PIN, HIGH, 25000);
  if (duration == 0) return 99.0;
  return duration * 0.034 / 2.0;
}

void stopMotors() {
  digitalWrite(IN1, LOW); digitalWrite(IN2, LOW);
  digitalWrite(IN3, LOW); digitalWrite(IN4, LOW);
  currentMotion = 'S';
}

void loop() {
  // 1. 讀取超音波感測器 -> 距離太近 -> 自動停止
  float dist = readUltrasonicDistance();
  Serial.print("DIST:");
  Serial.println(dist, 1);

  if (dist < SAFE_DISTANCE_CM && currentMotion == 'F') {
    stopMotors();
  }

  // 2. 接收 RPi 轉發之網頁指令 (對應講義 controlled.ino switch(c))
  if (Serial.available() > 0) {
    char c = Serial.read();
    switch (c) {
      case 'F': // 前進 (若距離大於安全門檻才允許前進)
        if (dist >= SAFE_DISTANCE_CM) {
          digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
          digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
          currentMotion = 'F';
        }
        break;
      case 'B': // 後退
        digitalWrite(IN1, LOW); digitalWrite(IN2, HIGH);
        digitalWrite(IN3, LOW); digitalWrite(IN4, HIGH);
        currentMotion = 'B';
        break;
      case 'L': // 左轉
        digitalWrite(IN1, LOW); digitalWrite(IN2, HIGH);
        digitalWrite(IN3, HIGH); digitalWrite(IN4, LOW);
        currentMotion = 'L';
        break;
      case 'R': // 右轉
        digitalWrite(IN1, HIGH); digitalWrite(IN2, LOW);
        digitalWrite(IN3, LOW); digitalWrite(IN4, HIGH);
        currentMotion = 'R';
        break;
      case 'S': // 手動停止
        stopMotors();
        break;
      case 'V': // 吸塵馬達啟動
        digitalWrite(VACUUM_PIN, HIGH);
        break;
      case 'v': // 吸塵馬達手動停止
        digitalWrite(VACUUM_PIN, LOW);
        break;
      case '1': myServo.write(45); break;
      case '2': myServo.write(90); break;
      case '3': myServo.write(135); break;
    }
  }
  delay(80);
}
